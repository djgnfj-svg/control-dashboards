def interpret(payload, fields):
    out = {}
    for f in fields:
        t, off = f['type'], f['offset']
        if t == 'u8':
            val = payload[off]
        else:
            val = struct.unpack(_TYPE_FMT[t],
                  payload[off:off + _TYPE_SIZE[t]])[0]
            if f.get('int'):   val = round(val)
            elif 'round' in f: val = round(val, f['round'])
        out[f['name']] = val
    return out

def try_allocate(store, mos, locks, positions, robot_mos,
                 online_ids=None, errors=None, reachable=None,
                 registered_ids=None):
    bucket = store.unassigned.get(mos)
    if not bucket:
        return None
    head = bucket[0]
    tx, ty = float(head.params.get("x", 0.0)), float(head.params.get("y", 0.0))
    node_index = head.params.get("node_index")
    candidate_ids = [rid for rid, m in robot_mos.items()
                     if m == mos and (registered_ids is None
                                      or rid in registered_ids)]
    chosen = _pick_nearest_available(tx, ty, candidate_ids, locks, positions,
                                     online_ids=online_ids, errors=errors,
                                     node_index=node_index, reachable=reachable)
    if chosen is None:
        return None
    bucket.pop(0)
    return head, chosen

def retry_tick(publish_fn) -> None:
    with state.CMD_ACK_MUTEX:
        for robot_id, by_cmd in list(state.ROBOT_ACK_PENDING.items()):
            for cmd, pending in list(by_cmd.items()):
                if pending.get('gave_up'):
                    since = pending.get('gave_up_at') or pending['sent_at']
                    if (now - datetime.fromisoformat(since)).total_seconds() \
                            >= GIVE_UP_KEEP_SECONDS:

                    continue
                pending['retries'] += 1
                to_retry.append((robot_id, pending['topic'], pending['payload'], cmd))

def lock_expired(lock: dict, now: datetime) -> bool:
    if not lock:
        return False
    if lock.get("source") == "user_call":
        return is_idle(lock, now)
    # manual 연결 기준 시각: 폴링(last_seen) > 승격(promoted_at) > 획득(acquired_at)
    if lock.get("last_seen"):
        conn_at, conn_limit = lock["last_seen"], MANUAL_CONN_TIMEOUT_SECONDS
    elif lock.get("promoted_at"):
        conn_at, conn_limit = lock["promoted_at"], ARRIVAL_HOLD_SECONDS
    else:
        conn_at, conn_limit = lock.get("acquired_at"), MANUAL_CONN_TIMEOUT_SECONDS
    if conn_at and (now - datetime.fromisoformat(conn_at)).total_seconds() > conn_limit:
        return True
    cands = []
    for k in ("last_command_at", "last_robot_active_at", "last_queue_hold_at"):
        v = lock.get(k)
        if v:
            cands.append(datetime.fromisoformat(v))
    if cands and (now - max(cands)).total_seconds() > MANUAL_IDLE_TIMEOUT_SECONDS:
        return True
    return False

참조 비교 diff

export function diffByKey(prev, next, keyFn) {
  const prevByKey = new Map(prev.map(r => [keyFn(r), r]));
  const nextKeys = new Set(), added = [], updated = [];
  for (const row of next) {
    const key = keyFn(row); nextKeys.add(key);
    const old = prevByKey.get(key);
    if (old === undefined) added.push(row);
    else if (old !== row)  updated.push(row); // 참조가 다르면 바뀐 것
  }
  const removed = [...prevByKey.keys()]
    .filter(k => !nextKeys.has(k));
  return { added, updated, removed };
}
